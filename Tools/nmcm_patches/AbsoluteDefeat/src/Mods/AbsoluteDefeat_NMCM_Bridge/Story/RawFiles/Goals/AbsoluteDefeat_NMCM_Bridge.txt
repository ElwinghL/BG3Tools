INITSECTION
// AbsoluteDefeat_NMCM_Bridge :: NMCM front end for Absolute Defeat's own MCM settings.
// Self contained. This goal OWNS DB_ADNB_Opt, DB_ADNB_Debug and DB_ADNB_Busy: no other goal writes
// to them. It subscribes to the framework by declaring a body of PROC_NMCM_Boot().
// TOP-LEVEL goal, with no ParentTargetEdge: with one, this goal is dead in a session that starts
// outside its parent act.
// Defaults below mirror Absolute Defeat's own MCM_blueprint.json ("Default" field of each setting):
// include_summons = false, downed_protection = true, debug_level = 0.
DB_ADNB_Opt("SUMMONS", 0);
DB_ADNB_Opt("DOWNED", 1);
DB_ADNB_Debug(0);
KBSECTION
// ===================================================================================
// Boot: claim the slot, seed/re-sync on every startup, enable this mod's own events.
// ===================================================================================
PROC
PROC_NMCM_Boot()
THEN
PROC_NMCM_Mark("NMCM_SLOT3_CLAIMED", 1);
PROC_NMCM_ClaimSlot(3, "AbsoluteDefeat_NMCM_Bridge");
// Seed before mirror: within one file, textual order of PROC bodies is the order they run in.
// Guarded PER FACT, not by a single version flag, so a later update that adds a new setting seeds
// only the new one and never touches what the player already chose.
PROC
PROC_NMCM_Boot()
AND
NOT DB_ADNB_Opt("SUMMONS", _)
THEN
DB_ADNB_Opt("SUMMONS", 0);
PROC
PROC_NMCM_Boot()
AND
NOT DB_ADNB_Opt("DOWNED", _)
THEN
DB_ADNB_Opt("DOWNED", 1);
PROC
PROC_NMCM_Boot()
AND
NOT DB_ADNB_Debug(_)
THEN
DB_ADNB_Debug(0);
// Mirror resync, both switches, POSITIVE half. Also covers a companion recruited later, who joins
// with none of the markers.
PROC
PROC_NMCM_Boot()
AND
DB_ADNB_Opt("SUMMONS", 1)
THEN
PROC_NMCM_Mark("ADNB_OPT_SUMMONS", 1);
PROC
PROC_NMCM_Boot()
AND
DB_ADNB_Opt("DOWNED", 1)
THEN
PROC_NMCM_Mark("ADNB_OPT_DOWNED", 1);
// Mirror resync, NEGATIVE half. Without this a marker stamped by an older session survives in the
// save and the box reads ticked while the DB says off. Removing a passive that is not there is a
// no-op, so this costs nothing.
PROC
PROC_NMCM_Boot()
AND
DB_ADNB_Opt("SUMMONS", 0)
THEN
PROC_NMCM_Mark("ADNB_OPT_SUMMONS", 0);
PROC
PROC_NMCM_Boot()
AND
DB_ADNB_Opt("DOWNED", 0)
THEN
PROC_NMCM_Mark("ADNB_OPT_DOWNED", 0);
// The debug level carrier GRANTS the action resource, so it is applied per player rather than
// through PROC_NMCM_Mark, which is for the passives the SCREEN READS.
PROC
PROC_NMCM_Boot()
AND
DB_Players(_P)
THEN
AddPassive(_P, "ADNB_DEBUG_CARRIER");
PROC
PROC_NMCM_Boot()
AND
DB_ADNB_Debug(_)
THEN
PROC_ADNB_DebugApply();
// Enable this mod's own UI events. WITHOUT THIS, CLICKS NEVER ARRIVE in a campaign already in
// progress, with nothing in any log: the set of enabled tutorial events is per character and is
// born with the character.
PROC
PROC_NMCM_Boot()
AND
DB_Players(_P)
THEN
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_OPT_SUMMONS_0ccdef93-b9d6-43ef-862d-72ec16f1c076);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_OPT_DOWNED_e6b5e30e-d429-4c4e-843a-c934102e1ae0);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_DEBUG_UP_daccd6e2-e48f-4a3f-ae33-6c66fc088540);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_DEBUG_DOWN_8ef58cd3-b611-446a-aab6-ef90805c365f);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_BTN_SURRENDER_e1f39073-e1ca-426b-9c9d-8448710ea2f5);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_BTN_SOFTLOCKFIX_0dabccf2-1f2f-492c-9d8b-9b7fcc9050eb);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_PAGE_OPENED_7475728c-6d81-43d6-9897-67db341aa391);
EnableTutorialEvent(_P, (TUTORIALEVENT)ADNB_ACT_RESET_90809479-679a-478d-b063-c9e745ae4567);
// Zero-arity signal the Lua bridge (BootstrapServer.lua) listens for via
// Ext.Osiris.RegisterListener("PROC_ADNB_RequestSync", 0, "after", ...): a good moment to pull the
// THREE real current values from Mods.BG3MCM.MCMAPI:GetSettingValue and push them back into this
// goal's own DB facts (Osi.PROC_ADNB_ApplySummons/ApplyDowned/ApplyDebug), so this page's display
// reflects Absolute Defeat's actual stored settings even when they were last changed from AD's own
// native MCM page rather than from here.
PROC
PROC_NMCM_Boot()
THEN
PROC_ADNB_RequestSync();
// The page re-announces the debug mirror on open: GetHostCharacter is undefined during
// LevelLoaded/SavegameLoaded, so PartyIncreaseActionResourceValue could not run there for a
// companion who was not yet a valid host. Re-applying here covers that case.
IF
TutorialEvent(_E, ADNB_PAGE_OPENED_7475728c-6d81-43d6-9897-67db341aa391)
THEN
PROC_ADNB_DebugApply();
// ===================================================================================
// Checkboxes: click -> "row was pressed" only. The goal owns the state and decides the new value.
// _McmId is the id Absolute Defeat's own MCM_blueprint.json declares for this setting, forwarded
// to the Lua bridge unchanged.
// ===================================================================================
IF
TutorialEvent(_E, ADNB_OPT_SUMMONS_0ccdef93-b9d6-43ef-862d-72ec16f1c076)
THEN
PROC_ADNB_Toggle("SUMMONS", "ADNB_OPT_SUMMONS", "include_summons");
IF
TutorialEvent(_E, ADNB_OPT_DOWNED_e6b5e30e-d429-4c4e-843a-c934102e1ae0)
THEN
PROC_ADNB_Toggle("DOWNED", "ADNB_OPT_DOWNED", "downed_protection");
// Generic toggle, in three bodies, guarded so both bodies do not fire on the same event.
// body 1: was OFF, turn it on.
PROC
PROC_ADNB_Toggle((STRING)_K, (STRING)_Pass, (STRING)_McmId)
AND
DB_ADNB_Opt(_K, 0)
AND
NOT DB_ADNB_Busy(1)
THEN
DB_ADNB_Busy(1);
NOT DB_ADNB_Opt(_K, 0);
DB_ADNB_Opt(_K, 1);
PROC_NMCM_Mark(_Pass, 1);
PROC_ADNB_SyncSetting(_McmId, 1);
// body 2: was ON, turn it off.
PROC
PROC_ADNB_Toggle((STRING)_K, (STRING)_Pass, (STRING)_McmId)
AND
DB_ADNB_Opt(_K, 1)
AND
NOT DB_ADNB_Busy(1)
THEN
DB_ADNB_Busy(1);
NOT DB_ADNB_Opt(_K, 1);
DB_ADNB_Opt(_K, 0);
PROC_NMCM_Mark(_Pass, 0);
PROC_ADNB_SyncSetting(_McmId, 0);
// body 3, LAST: release the guard.
PROC
PROC_ADNB_Toggle((STRING)_K, (STRING)_Pass, (STRING)_McmId)
AND
DB_ADNB_Busy(1)
THEN
NOT DB_ADNB_Busy(1);
// ===================================================================================
// Debug level stepper (Absolute Defeat "debug_level", slider_int, Options.Min=0, Options.Max=2).
// Same numeric-mirror law as NMCM's own reference control: the Action Resource is a MIRROR, never
// the owner, and it is mirrored WITHOUT reading it (floor with a large negative delta, which the
// engine clamps to zero, then add the value the DB owns).
// ===================================================================================
IF
TutorialEvent(_E, ADNB_DEBUG_UP_daccd6e2-e48f-4a3f-ae33-6c66fc088540)
THEN
PROC_ADNB_DebugStep(1);
IF
TutorialEvent(_E, ADNB_DEBUG_DOWN_8ef58cd3-b611-446a-aab6-ef90805c365f)
THEN
PROC_ADNB_DebugStep(-1);
// The range is enforced on the OWNER of the value (0..2, from MCM_blueprint.json Options.Min/Max),
// not on the resource, whose own clamp is unrelated.
PROC
PROC_ADNB_DebugStep((INTEGER)_D)
AND
DB_ADNB_Debug(_Old)
AND
IntegerSum(_Old, _D, _Raw)
AND
IntegerMax(_Raw, 0, _Floor)
AND
IntegerMin(_Floor, 2, _New)
THEN
PROC_ADNB_DebugSetTo(_New);
PROC
PROC_ADNB_DebugSetTo((INTEGER)_N)
AND
DB_ADNB_Debug(_Old)
THEN
NOT DB_ADNB_Debug(_Old);
DB_ADNB_Debug(_N);
PROC_ADNB_DebugApply();
PROC_ADNB_SyncSetting("debug_level", _N);
// The mirror: DB to Action Resource, without ever reading the resource. GetHostCharacter gives ONE
// character and the resource is party wide, so one call is enough.
PROC
PROC_ADNB_DebugApply()
AND
DB_ADNB_Debug(_N)
AND
GetHostCharacter(_H)
AND
IntegerToReal(_N, _R)
AND
RealSubtract(0.0, 1000.0, _Floor)
THEN
PartyIncreaseActionResourceValue(_H, "ADNB_DebugG", _Floor);
PartyIncreaseActionResourceValue(_H, "ADNB_DebugG", _R);
// ===================================================================================
// event_button rows: fire and forget, no state of our own. Forwarded to the Lua bridge, which
// calls the same MCM.EventButton callback Absolute Defeat's own page would have triggered.
// ===================================================================================
IF
TutorialEvent(_E, ADNB_BTN_SURRENDER_e1f39073-e1ca-426b-9c9d-8448710ea2f5)
THEN
PROC_ADNB_Fire("btn_surrender");
IF
TutorialEvent(_E, ADNB_BTN_SOFTLOCKFIX_0dabccf2-1f2f-492c-9d8b-9b7fcc9050eb)
THEN
PROC_ADNB_Fire("btn_softlockfix");
// ===================================================================================
// This mod's OWN reset, reached only by ADNB_ACT_RESET. Restores Absolute Defeat's own blueprint
// defaults and pushes them to MCM the same way a normal click would. Deliberately leaves
// NMCM_SLOT3_CLAIMED alone: claiming is not a player preference, and clearing it would make this
// mod vanish from the menu until the next startup.
// ===================================================================================
IF
TutorialEvent(_E, ADNB_ACT_RESET_90809479-679a-478d-b063-c9e745ae4567)
THEN
PROC_ADNB_Reset();
PROC
PROC_ADNB_Reset()
AND
DB_ADNB_Opt("SUMMONS", _S)
THEN
NOT DB_ADNB_Opt("SUMMONS", _S);
DB_ADNB_Opt("SUMMONS", 0);
PROC_NMCM_Mark("ADNB_OPT_SUMMONS", 0);
PROC_ADNB_SyncSetting("include_summons", 0);
PROC
PROC_ADNB_Reset()
AND
DB_ADNB_Opt("DOWNED", _D)
THEN
NOT DB_ADNB_Opt("DOWNED", _D);
DB_ADNB_Opt("DOWNED", 1);
PROC_NMCM_Mark("ADNB_OPT_DOWNED", 1);
PROC_ADNB_SyncSetting("downed_protection", 1);
PROC
PROC_ADNB_Reset()
AND
DB_ADNB_Debug(_)
THEN
PROC_ADNB_DebugSetTo(0);
// ===================================================================================
// Called FROM Lua (Osi.PROC_ADNB_ApplySummons/ApplyDowned/ApplyDebug) during the resync requested
// by PROC_ADNB_RequestSync above. Sets this goal's own DB fact and marker directly from Absolute
// Defeat's real current MCM value, WITHOUT forwarding back to PROC_ADNB_SyncSetting: the value
// already came from MCM, so echoing it back would be a needless round trip, not a feedback loop
// (the value is identical either way), but it is skipped on principle.
// ===================================================================================
PROC
PROC_ADNB_ApplySummons((INTEGER)_V)
AND
DB_ADNB_Opt("SUMMONS", _Old)
AND
_V != _Old
THEN
NOT DB_ADNB_Opt("SUMMONS", _Old);
DB_ADNB_Opt("SUMMONS", _V);
PROC_NMCM_Mark("ADNB_OPT_SUMMONS", _V);
PROC
PROC_ADNB_ApplyDowned((INTEGER)_V)
AND
DB_ADNB_Opt("DOWNED", _Old)
AND
_V != _Old
THEN
NOT DB_ADNB_Opt("DOWNED", _Old);
DB_ADNB_Opt("DOWNED", _V);
PROC_NMCM_Mark("ADNB_OPT_DOWNED", _V);
PROC
PROC_ADNB_ApplyDebug((INTEGER)_V)
AND
DB_ADNB_Debug(_Old)
AND
_V != _Old
THEN
NOT DB_ADNB_Debug(_Old);
DB_ADNB_Debug(_V);
PROC_ADNB_DebugApply();
EXITSECTION
ENDEXITSECTION
